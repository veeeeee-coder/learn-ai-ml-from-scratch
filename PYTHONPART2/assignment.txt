def print_even(a, b):
    for i in range(a, b + 1):
        if i % 2 == 0:
            print(i)

a = int(input("Enter first number: "))
b = int(input("Enter second number: "))

print_even(a, b)


def print_digits(n):
    while n > 0:
        digit = n % 10
        print(digit)
        n = n // 10

num = int(input("Enter a number: "))
print_digits(num)



def count_digits(n):
    count = 0

    while n > 0:
        count += 1
        n = n // 10

    return count

num = int(input("Enter a number: "))
print("Number of digits =", count_digits(num))


def sum_of_digits(n):
    total = 0

    while n > 0:
        total += n % 10
        n = n // 10

    return total

num = int(input("Enter a number: "))
print("Sum of digits =", sum_of_digits(num))

def calculator(a, b, operation):

    if operation == "+":
        return a + b

    elif operation == "-":
        return a - b

    elif operation == "*":
        return a * b

    elif operation == "/":
        return a / b

    else:
        return "Invalid operation"

a = float(input("Enter first number: "))
b = float(input("Enter second number: "))
op = input("Enter operation (+, -, *, /): ")

print("Result =", calculator(a, b, op))






def is_prime(n):

    if n < 2:
        return False

    for i in range(2, n):
        if n % i == 0:
            return False

    return True

num = int(input("Enter a number: "))

if is_prime(num):
    print("Prime Number")
else:
    print("Not a Prime Number")
