class Product:
count=0

def __init__(self,name,price):
self.name=name
self.price=price
Product.count+=1

def get_info(self):
print(f"price of {name} is {price}")

@classmethod
def get_count(cls):
print(f"{cls.count}")

@staticmethod
def get_final(price,discount):
print(f"discounted price={price-(price*discount/100)}")
